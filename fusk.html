Bookmarklet + console-script som kör direkt på badnewsswedish.eu

Steg 1 av 2 — enklaste sättet

Dra bookmarklet till bokmärkesfältet

Dra knappen nedan till din webbläsares bokmärkeslängd. Gå sedan till badnewsswedish.eu, starta spelet och klicka bokmärket.

Dra denna knapp till bokmärkeslängden ⚡ FUSK – DÅLIGA NYHETER ← dra hit → [bokmärken] fältet längst upp i webbläsaren

Om bokmärkeslängden inte syns: tryck Ctrl+Shift+B (Windows/Linux) eller Cmd+Shift+B (Mac).

Steg 2 — alternativt

Kör direkt i webbläsarkonsolen

Tryck F12 på badnewsswedish.eu, välj fliken Console, klistra in koden nedan och tryck Enter.

(function() {
  var FOLLOWERS = 99500;
  var CRED = 95;

  function findReactState(el) {
    var key = Object.keys(el).find(k => k.startsWith('__reactFiber') || k.startsWith('__reactInternalInstance'));
    return key ? el[key] : null;
  }

  function traverseFiber(fiber, depth) {
    if (!fiber || depth > 30) return;
    var s = fiber.memoizedState;
    while (s) {
      if (s.queue && s.queue.dispatch) { try { s.queue.dispatch(FOLLOWERS); } catch(e) {} }
      s = s.next;
    }
    traverseFiber(fiber.child, depth + 1);
    traverseFiber(fiber.sibling, depth + 1);
  }

  try {
    var els = document.querySelectorAll('[ng-controller],[data-ng-controller],[ng-app]');
    els.forEach(function(el) {
      try {
        var scope = window.angular.element(el).scope();
        if (!scope) return;
        if ('followers' in scope) scope.followers = FOLLOWERS;
        if ('credibility' in scope) scope.credibility = CRED;
        scope.$apply && scope.$apply();
      } catch(e) {}
    });
  } catch(e) {}

  ['#app','#game','main'].forEach(function(sel) {
    var el = document.querySelector(sel);
    if (!el) return;
    var vue = el.__vue__ || el.__vue_app__;
    if (vue) {
      try {
        var d = vue.$data;
        if (d && 'followers' in d) d.followers = FOLLOWERS;
        if (d && 'credibility' in d) d.credibility = CRED;
      } catch(e) {}
    }
  });

  ['game','gameState','state','store','vm','app'].forEach(function(n) {
    try {
      var obj = window[n];
      if (!obj) return;
      if ('followers' in obj) obj.followers = FOLLOWERS;
      if ('credibility' in obj) obj.credibility = CRED;
      if (obj.setState) obj.setState({ followers: FOLLOWERS, credibility: CRED });
      if (obj.commit) obj.commit('setFollowers', FOLLOWERS);
    } catch(e) {}
  });

  document.querySelectorAll('[id],[class]').forEach(function(el) {
    var fiber = findReactState(el);
    if (fiber) traverseFiber(fiber, 0);
  });

  document.querySelectorAll('*').forEach(function(el) {
    if (el.children.length > 0) return;
    var txt = el.textContent.trim();
    if (/^\d[\d\s.,]*$/.test(txt) && parseInt(txt.replace(/\D/g,'')) < 100000) {
      el.textContent = FOLLOWERS.toLocaleString('sv-SE');
    }
  });

  document.querySelectorAll('[class*="badge"],[class*="Badge"],[class*="medal"]').forEach(function(el) {
    el.style.filter = 'none'; el.style.opacity = '1'; el.style.background = '#ff0000';
  });

  document.querySelectorAll('[class*="locked"],[class*="inactive"],[class*="disabled"]').forEach(function(el) {
    el.classList.remove('locked','inactive','disabled');
    el.classList.add('earned','active');
    el.style.filter = 'none'; el.style.opacity = '1';
  });

  var existing = document.getElementById('__fusk_overlay__');
  if (existing) { existing.remove(); return; }

  var overlay = document.createElement('div');
  overlay.id = '__fusk_overlay__';
  overlay.style.cssText = 'position:fixed;bottom:0;right:0;z-index:2147483647;pointer-events:none;display:flex;align-items:flex-end;justify-content:flex-end;padding:20px';

  var panel = document.createElement('div');
  panel.style.cssText = 'pointer-events:all;background:#ffff00;color:#000;font-family:monospace;font-size:12px;padding:16px 20px;border:3px solid #000;min-width:220px;box-shadow:0 4px 20px rgba(0,0,0,0.3)';
  panel.innerHTML = '
FUSK AKTIVERAT
' + FOLLOWERS.toLocaleString('sv-SE') + '
FÖLJARE
TROVÄRDIGHET' + CRED + '%
🎭😱⚔️🔍🗑️👾
';

  overlay.appendChild(panel);
  document.body.appendChild(overlay);
  console.log('%c⚡ FUSK AKTIVERAT','background:#ffff00;color:#000;font-size:16px;font-weight:bold;padding:6px 14px');
})();

Vad gör scriptet?

  • React / Angular / VueFörsöker hitta och modifiera spelets interna JavaScript-tillstånd direkt
  • Globala variablerSöker igenom vanliga globala state-variabler (game, gameState, store…)
  • DOM-manipulationÄndrar synliga siffror på sidan och aktiverar låsta badge-element
  • AUTO-SVARKlicka "AUTO-SVAR" i panelen för att automatiskt välja det längsta/mest aggressiva svaret
  • Overlay-panelVisar panel med 99 500 följare, 95% trovärdighet och alla 6 märken

Steg-för-steg

Bookmarklet:
1. Dra den gula knappen till bokmärkeslängden
2. Gå till badnewsswedish.eu
3. Starta spelet och kom till en fråga
4. Klicka bokmärket

Console:
1. Gå till badnewsswedish.eu
2. Tryck F12 → Console
3. Kopiera och klistra in koden
4. Tryck Enter

Obs: Spelets slutresultat kan skilja sig från det visade om poengräkning sker server-side. Använd AUTO-SVAR för att snabbt spela igenom frågorna med de bästa svaren.
