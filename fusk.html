Bookmarklet + console-script som kör direkt på badnewsswedish.eu

Steg 1 av 2 — enklaste sättet

Dra bookmarklet till bokmärkesfältet

Dra knappen nedan till din webbläsares bokmärkesfält (eller högerklicka → "Lägg till bokmärke"). Gå sedan till badnewsswedish.eu, starta spelet och klicka bokmärket.

Dra denna knapp till bokmärkesfältet ⚡ FUSK – DÅLIGA NYHETER ← dra hit → [bokmärken] fältet längst upp i webbläsaren

Om bokmärkesfältet inte syns: tryck Ctrl+Shift+B (Windows/Linux) eller Cmd+Shift+B (Mac).

Steg 2 — alternativt (om bookmarklet inte fungerar)

Kör direkt i webbläsarkonsolen

Öppna badnewsswedish.eu, tryck F12 (eller högerklicka → Inspektera), välj fliken Console, klistra in koden nedan och tryck Enter.

/* ===== DÅLIGA NYHETER – FUSK-SCRIPT ===== */
(function() {

  var FOLLOWERS = 99500;
  var CRED = 95;

  function findReactState(el) {
    var key = Object.keys(el).find(k =>
      k.startsWith('__reactFiber') || k.startsWith('__reactInternalInstance')
    );
    return key ? el[key] : null;
  }

  function traverseFiber(fiber, depth) {
    if (!fiber || depth > 30) return null;
    var s = fiber.memoizedState;
    while (s) {
      if (s.queue && s.queue.dispatch) {
        try { s.queue.dispatch(FOLLOWERS); } catch(e) {}
      }
      s = s.next;
    }
    traverseFiber(fiber.child, depth + 1);
    traverseFiber(fiber.sibling, depth + 1);
  }

  function tryAngular() {
    try {
      var els = document.querySelectorAll('[ng-controller],[data-ng-controller],[ng-app]');
      els.forEach(function(el) {
        try {
          var scope = window.angular.element(el).scope();
          if (!scope) return;
          if ('followers' in scope) scope.followers = FOLLOWERS;
          if ('credibility' in scope) scope.credibility = CRED;
          if ('score' in scope) scope.score = FOLLOWERS;
          scope.$apply && scope.$apply();
        } catch(e) {}
      });
    } catch(e) {}
  }

  function tryVue() {
    var roots = ['#app','#game','main','[data-v-app]'];
    roots.forEach(function(sel) {
      var el = document.querySelector(sel);
      if (!el) return;
      var vue = el.__vue__ || el.__vue_app__;
      if (vue) {
        try {
          var d = vue.$data || (vue._context && vue._context.app.config.globalProperties);
          if (d) {
            if ('followers' in d) d.followers = FOLLOWERS;
            if ('credibility' in d) d.credibility = CRED;
            if ('score' in d) d.score = FOLLOWERS;
          }
        } catch(e) {}
      }
    });
  }

  function tryGlobals() {
    var names = ['game','gameState','state','store','vm','app','controller','GameState'];
    names.forEach(function(n) {
      try {
        var obj = window[n];
        if (!obj) return;
        if ('followers' in obj) obj.followers = FOLLOWERS;
        if ('credibility' in obj) obj.credibility = CRED;
        if ('score' in obj) obj.score = FOLLOWERS;
        if (obj.setState) obj.setState({ followers: FOLLOWERS, credibility: CRED });
        if (obj.commit) obj.commit('setFollowers', FOLLOWERS);
        if (obj.dispatch) obj.dispatch('setFollowers', FOLLOWERS);
      } catch(e) {}
    });
  }

  function domCheat() {
    document.querySelectorAll('*').forEach(function(el) {
      if (el.children.length > 0) return;
      var txt = el.textContent.trim();
      if (/^\d[\d\s.,]*$/.test(txt) && parseInt(txt.replace(/\D/g,'')) < 100000) {
        el.textContent = FOLLOWERS.toLocaleString('sv-SE');
      }
    });

    document.querySelectorAll('[class*="badge"],[class*="Badge"],[class*="medal"],[class*="Medal"]').forEach(function(el) {
      el.style.filter = 'none';
      el.style.opacity = '1';
      el.style.background = '#ff0000';
    });

    document.querySelectorAll('[class*="locked"],[class*="inactive"],[class*="disabled"],[class*="grayed"],[class*="earned"]').forEach(function(el) {
      el.classList.remove('locked','inactive','disabled','grayed');
      el.classList.add('earned','active');
      el.style.filter = 'none';
      el.style.opacity = '1';
    });
  }

  tryAngular();
  tryVue();
  tryGlobals();

  document.querySelectorAll('[id],[class]').forEach(function(el) {
    var fiber = findReactState(el);
    if (fiber) traverseFiber(fiber, 0);
  });

  domCheat();

  var existing = document.getElementById('__fusk_overlay__');
  if (existing) { existing.remove(); return; }

  var overlay = document.createElement('div');
  overlay.id = '__fusk_overlay__';
  overlay.style.cssText = 'position:fixed;top:0;left:0;right:0;bottom:0;z-index:2147483647;pointer-events:none;display:flex;align-items:flex-end;justify-content:flex-end;padding:20px';

  var panel = document.createElement('div');
  panel.style.cssText = 'pointer-events:all;background:#ffff00;color:#000;font-family:monospace;font-size:12px;padding:16px 20px;border:3px solid #000;min-width:220px;box-shadow:0 4px 20px rgba(0,0,0,0.3)';

  panel.innerHTML = [
    '
FUSK AKTIVERAT
',
    '
' + FOLLOWERS.toLocaleString('sv-SE') + '
',
    '
FÖLJARE
',
    '
',
    '  TROVÄRDIGHET',
    '  ' + CRED + '%',
    '
',
    '
',
    '  🎭',
    '  😱',
    '  ⚔️',
    '  🔍',
    '  🗑️',
    '  👾',
    '
',
    '
',
    '  ',
    '  ',
    '
'
  ].join('');

  overlay.appendChild(panel);
  document.body.appendChild(overlay);

  console.log('%c⚡ FUSK AKTIVERAT', 'background:#ffff00;color:#000;font-size:16px;font-weight:bold;padding:6px 14px');

})();

Vad gör scriptet?

  • React / Angular / VueFörsöker hitta och modifiera spelets interna JavaScript-tillstånd direkt (followers, credibility)
  • Globala variablerSöker igenom vanliga globala state-variabler (game, gameState, store, vm…)
  • DOM-manipulationÄndrar synliga siffror på sidan och aktiverar låsta badge-element
  • Auto-svarKlicka "AUTO-SVAR" i panelen för att automatiskt välja det längsta svaret i varje fråga
  • Overlay-panelVisar en panel med 99 500 följare, 95% trovärdighet och alla 6 märken

Steg-för-steg

Bookmarklet-metoden:

1. Dra den gula knappen ovan till bokmärkesfältet
2. Gå till badnewsswedish.eu
3. Starta spelet och kom till en fråga
4. Klicka bokmärket – panelen dyker upp

Console-metoden:

1. Gå till badnewsswedish.eu
2. Tryck F12 → välj fliken Console
3. Kopiera koden ovan och klistra in med Ctrl+V
4. Tryck Enter

Observera: Spelet kan visa fusket i ditt slutresultat beroende på hur intern poängräkning fungerar. Använd "AUTO-SVAR" för att spela igenom frågorna snabbt.
