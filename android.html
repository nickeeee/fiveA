Fuska på Android

Tre metoder – välj den som passar dig bäst

⭐ Enklast – rekommenderas

Firefox för Android stödjer riktiga webbläsar-extensions. Installera Tampermonkey och lägg in scriptet – det körs automatiskt varje gång du besöker badnewsswedish.eu.

1

Installera Firefox för Android

Sök efter "Firefox" i Google Play och installera. Öppna Firefox när det är klart.

2

Aktivera tillägg i Firefox

Tryck på menyn (⋮) längst ned till höger → Tillägg → Hantera tillägg. Tryck på + bredvid Tampermonkey och installera.

3

Öppna Tampermonkey Dashboard

Tryck på pusselbiten 🧩 i adressfältet → Tampermonkey → Dashboard → + (nytt script).

4

Klistra in scriptet

Ta bort all befintlig text i editorn och klistra in hela koden nedan. Tryck sedan Spara (disketten 💾).

// ==UserScript==
// @name         Dåliga Nyheter – Fusk
// @namespace    https://badnewsswedish.eu
// @version      1.0
// @description  Fuska på Dåliga Nyheter – maxpoäng & trovärdighet
// @match        https://www.badnewsswedish.eu/*
// @match        https://badnewsswedish.eu/*
// @grant        none
// @run-at       document-idle
// ==/UserScript==

(function () {
  'use strict';

  var FOLLOWERS = 99500;
  var CRED = 95;

  // Vänta tills spelet laddas
  function waitAndInject() {
    var attempts = 0;
    var interval = setInterval(function () {
      attempts++;
      if (attempts > 60) { clearInterval(interval); return; }

      // Kolla om spelet har laddats
      var gameEl = document.querySelector('[class*="game"], [class*="Game"], main, #app, #root');
      if (!gameEl) return;
      clearInterval(interval);
      injectCheatPanel();
      runCheats();
    }, 500);
  }

  function runCheats() {
    // Angular
    try {
      document.querySelectorAll('[ng-controller],[data-ng-controller],[ng-app]').forEach(function (el) {
        try {
          var scope = window.angular.element(el).scope();
          if (!scope) return;
          if ('followers' in scope) scope.followers = FOLLOWERS;
          if ('credibility' in scope) scope.credibility = CRED;
          if ('score' in scope) scope.score = FOLLOWERS;
          scope.$apply && scope.$apply();
        } catch (e) {}
      });
    } catch (e) {}

    // Vue
    ['#app', '#game', '#root', 'main'].forEach(function (sel) {
      var el = document.querySelector(sel);
      if (!el) return;
      var vue = el.__vue__ || el.__vue_app__;
      if (!vue) return;
      try {
        var d = vue.$data;
        if (!d) return;
        if ('followers' in d) d.followers = FOLLOWERS;
        if ('credibility' in d) d.credibility = CRED;
        if ('score' in d) d.score = FOLLOWERS;
      } catch (e) {}
    });

    // Globala variabler
    ['game', 'gameState', 'state', 'store', 'vm', 'app', 'controller'].forEach(function (n) {
      try {
        var obj = window[n];
        if (!obj) return;
        if ('followers' in obj) obj.followers = FOLLOWERS;
        if ('credibility' in obj) obj.credibility = CRED;
        if (obj.setState) obj.setState({ followers: FOLLOWERS, credibility: CRED });
        if (obj.commit) obj.commit('setFollowers', FOLLOWERS);
      } catch (e) {}
    });

    // React fiber
    document.querySelectorAll('[id],[class]').forEach(function (el) {
      var key = Object.keys(el).find(function (k) {
        return k.startsWith('__reactFiber') || k.startsWith('__reactInternalInstance');
      });
      if (!key) return;
      traverseFiber(el[key], 0);
    });

    // DOM: ändra synliga nummer
    document.querySelectorAll('*').forEach(function (el) {
      if (el.children.length > 0) return;
      var txt = el.textContent.trim();
      if (/^\d[\d\s.,]*$/.test(txt) && parseInt(txt.replace(/\D/g, '')) < 100000) {
        el.textContent = FOLLOWERS.toLocaleString('sv-SE');
      }
    });

    // Lås upp badges
    document.querySelectorAll('[class*="badge"],[class*="Badge"],[class*="medal"]').forEach(function (el) {
      el.style.filter = 'none';
      el.style.opacity = '1';
      el.style.background = '#ff0000';
    });
    document.querySelectorAll('[class*="locked"],[class*="inactive"],[class*="disabled"]').forEach(function (el) {
      el.classList.remove('locked', 'inactive', 'disabled');
      el.classList.add('earned', 'active');
      el.style.filter = 'none';
      el.style.opacity = '1';
    });
  }

  function traverseFiber(fiber, depth) {
    if (!fiber || depth > 30) return;
    var s = fiber.memoizedState;
    while (s) {
      if (s.queue && s.queue.dispatch) {
        try { s.queue.dispatch(FOLLOWERS); } catch (e) {}
      }
      s = s.next;
    }
    traverseFiber(fiber.child, depth + 1);
    traverseFiber(fiber.sibling, depth + 1);
  }

  function autoAnswer() {
    var choices = Array.from(document.querySelectorAll(
      'button:not([disabled]), [class*="choice"], [class*="Choice"], [class*="option"], [class*="answer"]'
    )).filter(function (el) {
      return el.offsetParent !== null && el.textContent.trim().length > 10;
    });
    if (choices.length < 1) { alert('Inga svar hittades. Är du på en fråga?'); return; }
    // Välj det längsta alternativet
    var best = choices.reduce(function (a, b) {
      return b.textContent.length > a.textContent.length ? b : a;
    });
    best.click();
  }

  function injectCheatPanel() {
    if (document.getElementById('__tm_fusk__')) return;

    var panel = document.createElement('div');
    panel.id = '__tm_fusk__';
    panel.style.cssText = [
      'position:fixed', 'bottom:16px', 'right:16px', 'z-index:2147483647',
      'background:#ffff00', 'color:#000',
      'font-family:monospace', 'font-size:13px',
      'padding:16px 18px',
      'border:3px solid #000',
      'min-width:200px',
      'box-shadow:0 4px 24px rgba(0,0,0,0.35)',
      'touch-action:none'
    ].join(';');

    panel.innerHTML = [
      '
⚡ FUSK AKTIVERAT
',
      '
' + FOLLOWERS.toLocaleString('sv-SE') + '
',
      '
FÖLJARE
',
      '
',
      '  TROVÄRDIGHET',
      '  ' + CRED + '%',
      '
',
      '
',
      '  🎭',
      '  😱',
      '  ⚔️',
      '  🔍',
      '  🗑️',
      '  👾',
      '
',
      '
',
      '  ',
      '  ',
      '  ',
      '
'
    ].join('');

    document.body.appendChild(panel);

    document.getElementById('__tm_cheat_btn__').addEventListener('click', function () { runCheats(); });
    document.getElementById('__tm_auto_btn__').addEventListener('click', function () { autoAnswer(); });
    document.getElementById('__tm_close_btn__').addEventListener('click', function () { panel.remove(); });

    // Gör panelen draggbar (touch)
    var dragging = false, sx, sy, ox = 16, oy = 16;
    panel.addEventListener('touchstart', function (e) {
      if (e.target.tagName === 'BUTTON') return;
      dragging = true;
      sx = e.touches[0].clientX;
      sy = e.touches[0].clientY;
    }, { passive: true });
    document.addEventListener('touchmove', function (e) {
      if (!dragging) return;
      var dx = e.touches[0].clientX - sx;
      var dy = e.touches[0].clientY - sy;
      ox = Math.max(0, Math.min(window.innerWidth - panel.offsetWidth, ox - dx));
      oy = Math.max(0, Math.min(window.innerHeight - panel.offsetHeight, oy - dy));
      panel.style.right  = ox + 'px';
      panel.style.bottom = oy + 'px';
      panel.style.left = 'auto';
      panel.style.top  = 'auto';
      sx = e.touches[0].clientX;
      sy = e.touches[0].clientY;
    }, { passive: true });
    document.addEventListener('touchend', function () { dragging = false; });
  }

  waitAndInject();

})();
5

Öppna badnewsswedish.eu i Firefox

Scriptet körs automatiskt. En gul panel dyker upp med:
• 🔁 FUSK IGEN – kör fusket på nytt
• ⚡ AUTO-SVAR – klickar bästa svaret automatiskt
• Du kan dra panelen med fingret

💡 Panelen är draggbar – håll och dra den till en bra position på skärmen.
🥝 Bra alternativ

Kiwi Browser är en Chromium-baserad webbläsare för Android som stödjer Chrome-extensions, inklusive Tampermonkey. Samma script som Firefox-metoden.

1

Installera Kiwi Browser

Sök efter "Kiwi Browser" i Google Play Store och installera.

2

Installera Tampermonkey

Öppna Kiwi Browser. Gå till Chrome Web Store:
chrome.google.com/webstore
Sök efter Tampermonkey och installera.

3

Lägg till scriptet

Tryck på menyn (⋮) → Extensions → Tampermonkey → Dashboard → + (nytt script).

Kopiera scriptet från Firefox-fliken ovan och klistra in. Spara.

4

Öppna spelet

Gå till badnewsswedish.eu i Kiwi Browser. Panelen dyker upp automatiskt.

📎 Fungerar i Chrome – kräver lite mer

Chrome på Android stödjer inte extensions men du kan lägga in en bookmarklet manuellt. Det är lite krångligare men fungerar.

1

Bokmärk valfri sida

Öppna Chrome → gå till valfri sida → tryck stjärnan ☆ i adressfältet (eller ⋮ → Bokmärk). Namnge det "⚡ Fusk".

2

Kopiera bookmarklet-koden

Kopiera hela koden nedan – det är ett komprimerat javascript:-URL.

3

Redigera bokmärket

Tryck på ⋮ i Chrome → Bokmärken → håll på ⚡ Fusk → Redigera bokmärke.

Radera allt i URL-fältet och klistra in koden du kopierade. Spara.

4

Använd bokmärket

Gå till badnewsswedish.eu. Starta spelet. Skriv ⚡ i adressfältet → välj bokmärket ⚡ Fusk ur listan. Panelen dyker upp.

💡 Tips: I Chrome Android räcker det att skriva de första bokstäverna av bokmärkets namn i adressfältet för att det ska dyka upp i förslag.
